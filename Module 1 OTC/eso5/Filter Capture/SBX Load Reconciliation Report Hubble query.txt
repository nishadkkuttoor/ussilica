SELECT SBXLOADDETAIL.SHDOCO                                                                             SBXLOADDETAIL_SHDOCO,
       ( TO_CHAR(SBXLOADDETAIL.ORDATE, N'YYYY') - 1900 ) * 1000 + TO_CHAR(SBXLOADDETAIL.ORDATE, N'DDD') SBXLOADDETAIL_ORDATE,
       ( TO_CHAR(SBXLOADDETAIL.GLDATE, N'YYYY') - 1900 ) * 1000 + TO_CHAR(SBXLOADDETAIL.GLDATE, N'DDD') SBXLOADDETAIL_GLDATE,
       SBXLOADDETAIL.SDDOC                                                                              SBXLOADDETAIL_SDDOC,
       SBXLOADDETAIL.SHMCU                                                                              SBXLOADDETAIL_SHMCU,
       SBXLOADDETAIL.SDLTTR                                                                             SBXLOADDETAIL_SDLTTR,
       SBXLOADDETAIL.SDNXTR                                                                             SBXLOADDETAIL_SDNXTR,
       SBXLOADDETAIL.SHAN8                                                                              SBXLOADDETAIL_SHAN8,
       SBXLOADDETAIL.SHSHAN                                                                             SBXLOADDETAIL_SHSHAN,
       F554201T.QCDS50                                                                                  F554201T_QCDS50,
       SBXLOADDETAIL.LOFA                                                                               SBXLOADDETAIL_LOFA,
       SBXLOADDETAIL.SHCARS                                                                             SBXLOADDETAIL_SHCARS,
       SBXLOADDETAIL.PROP                                                                               SBXLOADDETAIL_PROP,
       SBXLOADDETAIL.SHVR01                                                                             SBXLOADDETAIL_SHVR01,
       F554201T.QCLGL1                                                                                  F554201T_QCLGL1,
       F554201T.QCLGL2                                                                                  F554201T_QCLGL2,
       F554201T.QCLGL3                                                                                  F554201T_QCLGL3,
       F554201T.QCFSTR3                                                                                 F554201T_QCFSTR3,
       SBXLOADDETAIL.SANDTKT                                                                            SBXLOADDETAIL_SANDTKT,
       SBXLOADDETAIL.BOL                                                                                SBXLOADDETAIL_BOL,
       SBXLOADDETAIL.LOFADET                                                                            SBXLOADDETAIL_LOFADET,
       SBXLOADDETAIL.WELLDET                                                                            SBXLOADDETAIL_WELLDET,
       SBXLOADDETAIL.SANDWEIGHT                                                                         SBXLOADDETAIL_SANDWEIGHT,
       SBXLOADDETAIL.EXTWEIGHT                                                                          SBXLOADDETAIL_EXTWEIGHT,
       ( CAST(SBXLOADDETAIL.SANDWEIGHT AS FLOAT) / CAST(2000 AS FLOAT) )                                ID_CUSTOM_89d35e8e9f91315,
       ( CAST(SBXLOADDETAIL.EXTWEIGHT AS FLOAT) / CAST(2000 AS FLOAT) )                                 ID_CUSTOM_8a7616df9282d49,
       SBXLOADDETAIL.MILES                                                                              SBXLOADDETAIL_MILES,
       SBXLOADDETAIL.FRTAMT                                                                             SBXLOADDETAIL_FRTAMT,
       SBXLOADDETAIL.SANDAMT                                                                            SBXLOADDETAIL_SANDAMT,
       SBXLOADDETAIL.HOLAMT                                                                             SBXLOADDETAIL_HOLAMT,
       SBXLOADDETAIL.EXTAMT                                                                             SBXLOADDETAIL_EXTAMT,
       MAX(FLOOR(TO_NUMBER(NULL)))                                                                      ReportColumn1
FROM   (SELECT SHKCOO,
               SHDOCO,
               SHDCTO,
               SHVR01,
               SHAN8,
               SHSHAN,
               SHCARS,
               SHMCU,
               ORDATE,
               SANDWEIGHT,
               EXTWEIGHT,
               PROP,
               LOFA,
               MILES,
               BOL,
               SANDTKT,
               LOFADET,
               WELLDET,
               LOFADETPP,
               WELLDETPP,
               LOFADETPB,
               WELLDETPB,
               EXTAMT,
               LOFAAMT,
               WELLAMT,
               LOFAPPAMT,
               WELLPPAMT,
               LOFAPBAMT,
               WELLPBAMT,
               FRTAMT,
               FSCAMT,
               SANDAMT,
               HOLAMT,
               GLDATE,
               ( CASE
                   WHEN MXLTTR = 980
                        and MILTTR = 980 THEN MXLTTR
                   ELSE (select MAX(B.sdlttr)
                         from   proddta.F4211 B
                         where  B.SDDOCO = SHDOCO
                                and B.SDDCTO = SHDCTO
                                and B.SDKCOO = SHKCOO
                                and B.SDLITM <> 'HOLADD'
                                and B.SDAEXP <> 0)
                 END ) SDLTTR,
               SDNXTR,
               SDDOC
        FROM   (SELECT SHKCOO,
                       SHDOCO,
                       SHDCTO,
                       SHVR01,
                       SHAN8,
                       SHSHAN,
                       SHCARS,
                       SHMCU,
                       TO_DATE(TO_CHAR(SHTRDJ + 1900000), 'YYYYDDD')                     AS ORDATE,
                       SUM(DECODE(F4211.SDPRP1, 'COM', F4211.SDUORG / 1000))             AS SANDWEIGHT,
                       SUM(DECODE(F4211.SDPRP1, 'FRT', F4211.SDITWT / 10000))            AS EXTWEIGHT,
                       MAX(DECODE(F4211.SDPRP1, 'COM', F4211.SDDSC1))                    AS PROP,
                       MAX(DECODE(F4211.SDPRP1, 'COM', F4211.SDVEND))                    AS LOFA,
                       SUM(DECODE(TRIM(F4211.SDLITM), 'FRT', F4211.SDUORG / 1000))       AS MILES,
                       MAX(DECODE(TRIM(F4211.SDLITM), 'BOL', F4211.SDDSC1))              AS BOL,
                       MAX(DECODE(TRIM(F4211.SDLITM), 'SANDTKTNBR', F4211.SDDSC1))       AS SANDTKT,
                       SUM(DECODE(TRIM(F4211.SDLITM), 'LOFADET', F4211.SDUORG / 1000))   AS LOFADET,
                       SUM(DECODE(TRIM(F4211.SDLITM), 'WELLDET', F4211.SDUORG / 1000))   AS WELLDET,
                       SUM(DECODE(TRIM(F4211.SDLITM), 'LOFADETPP', F4211.SDUORG / 1000)) AS LOFADETPP,
                       SUM(DECODE(TRIM(F4211.SDLITM), 'WELLDETPP', F4211.SDUORG / 1000)) AS WELLDETPP,
                       SUM(DECODE(TRIM(F4211.SDLITM), 'LOFADETPB', F4211.SDUORG / 1000)) AS LOFADETPB,
                       SUM(DECODE(TRIM(F4211.SDLITM), 'WELLDETPB', F4211.SDUORG / 1000)) AS WELLDETPB,
                       SUM(F4211.SDAEXP / 1)                                             AS EXTAMT,
                       SUM(DECODE(TRIM(F4211.SDLITM), 'LOFADET', F4211.SDAEXP / 1))      AS LOFAAMT,
                       SUM(DECODE(TRIM(F4211.SDLITM), 'WELLDET', F4211.SDAEXP / 1))      AS WELLAMT,
                       SUM(DECODE(TRIM(F4211.SDLITM), 'LOFADETPP', F4211.SDAEXP / 1))    AS LOFAPPAMT,
                       SUM(DECODE(TRIM(F4211.SDLITM), 'WELLDETPP', F4211.SDAEXP / 1))    AS WELLPPAMT,
                       SUM(DECODE(TRIM(F4211.SDLITM), 'LOFADETPB', F4211.SDAEXP / 1))    AS LOFAPBAMT,
                       SUM(DECODE(TRIM(F4211.SDLITM), 'WELLDETPB', F4211.SDAEXP / 1))    AS WELLPBAMT,
                       SUM(DECODE(TRIM(F4211.SDPRP1), 'FRT', F4211.SDAEXP / 1))          AS FRTAMT,
                       SUM(DECODE(TRIM(F4211.SDLITM), 'FSC', F4211.SDAEXP / 1))          AS FSCAMT,
                       SUM(DECODE(TRIM(F4211.SDSRP1), '352', F4211.SDAEXP / 1))          AS SANDAMT,
                       SUM(DECODE(TRIM(F4211.SDLITM), 'HOLADD', F4211.SDAEXP / 1))       AS HOLAMT,
                       MAX(CASE
                             WHEN F4211.SDDGL = 0 THEN TO_DATE ('1900001', 'YYYYDDD')
                             WHEN F4211.SDDGL IS NULL THEN TO_DATE('1900001', 'YYYYDDD')
                             ELSE TO_DATE(TO_CHAR(F4211.SDDGL + 1900000), 'YYYYDDD')
                           END)                                                          AS GLDATE,
                       MAX(F4211.SDLTTR)                                                 AS MXLTTR,
                       MIN (F4211.SDLTTR)                                                AS MILTTR,
                       MAX(F4211.SDNXTR)                                                 AS SDNXTR,
                       MAX(F4211.SDDOC)                                                  AS SDDOC
                FROM   PRODDTA.F4201
                       INNER JOIN PRODDTA.F4211
                         ON F4201.SHDOCO = F4211.SDDOCO
                            AND F4201.SHDCTO = F4211.SDDCTO
                            AND F4201.SHKCOO = F4211.SDKCOO
                WHERE  F4201.SHDCTO = 'SX'
                GROUP  BY SHKCOO,
                          SHDOCO,
                          SHDCTO,
                          SHMCU,
                          SHVR01,
                          SHAN8,
                          SHSHAN,
                          SHCARS,
                          SHTRDJ)) SBXLOADDETAIL
       LEFT JOIN PRODDTA.F554201T F554201T
         ON ( ( SUBSTR(N'00000'
                        || TO_CHAR(RTRIM(LTRIM(SBXLOADDETAIL.SHKCOO))), LENGTH(TO_CHAR(RTRIM(LTRIM(SBXLOADDETAIL.SHKCOO)))) + 1, 5) = F554201T.QCKCOO )
              AND ( SBXLOADDETAIL.SHDOCO = F554201T.QCDOCO ) )
            AND ( SBXLOADDETAIL.SHDCTO = F554201T.QCDCTO )
WHERE  ( (( (( SBXLOADDETAIL.SHDOCO IN ( 22815083 ) ))
            AND (( NOT ( SBXLOADDETAIL.SDLTTR IN ( N'980' ) ) )) ))
         AND (( ( TO_CHAR(SBXLOADDETAIL.ORDATE, N'YYYY') - 1900 ) * 1000 + TO_CHAR(SBXLOADDETAIL.ORDATE, N'DDD') BETWEEN 125121 AND 125151 )) )
GROUP  BY SBXLOADDETAIL.SHDOCO,
          ( TO_CHAR(SBXLOADDETAIL.ORDATE, N'YYYY') - 1900 ) * 1000 + TO_CHAR(SBXLOADDETAIL.ORDATE, N'DDD'),
          ( TO_CHAR(SBXLOADDETAIL.GLDATE, N'YYYY') - 1900 ) * 1000 + TO_CHAR(SBXLOADDETAIL.GLDATE, N'DDD'),
          SBXLOADDETAIL.SDDOC,
          SBXLOADDETAIL.SHMCU,
          SBXLOADDETAIL.SDLTTR,
          SBXLOADDETAIL.SDNXTR,
          SBXLOADDETAIL.SHAN8,
          SBXLOADDETAIL.SHSHAN,
          F554201T.QCDS50,
          SBXLOADDETAIL.LOFA,
          SBXLOADDETAIL.SHCARS,
          SBXLOADDETAIL.PROP,
          SBXLOADDETAIL.SHVR01,
          F554201T.QCLGL1,
          F554201T.QCLGL2,
          F554201T.QCLGL3,
          F554201T.QCFSTR3,
          SBXLOADDETAIL.SANDTKT,
          SBXLOADDETAIL.BOL,
          SBXLOADDETAIL.LOFADET,
          SBXLOADDETAIL.WELLDET,
          SBXLOADDETAIL.SANDWEIGHT,
          SBXLOADDETAIL.EXTWEIGHT,
          ( CAST(SBXLOADDETAIL.SANDWEIGHT AS FLOAT) / CAST(2000 AS FLOAT) ),
          ( CAST(SBXLOADDETAIL.EXTWEIGHT AS FLOAT) / CAST(2000 AS FLOAT) ),
          SBXLOADDETAIL.MILES,
          SBXLOADDETAIL.FRTAMT,
          SBXLOADDETAIL.SANDAMT,
          SBXLOADDETAIL.HOLAMT,
          SBXLOADDETAIL.EXTAMT
ORDER  BY SBXLOADDETAIL_SHAN8 ASC,
          SBXLOADDETAIL_SHSHAN ASC,
          SBXLOADDETAIL_LOFA ASC,
          SBXLOADDETAIL_ORDATE ASC,
          SBXLOADDETAIL_SHVR01 ASC,
          SBXLOADDETAIL_SDLTTR ASC,
          SBXLOADDETAIL_SHCARS ASC,
          SBXLOADDETAIL_BOL ASC,
          SBXLOADDETAIL_LOFADET ASC,
          SBXLOADDETAIL_MILES ASC,
          SBXLOADDETAIL_SANDTKT ASC,
          SBXLOADDETAIL_WELLDET ASC,
          SBXLOADDETAIL_PROP ASC,
          SBXLOADDETAIL_SHDOCO ASC,
          SBXLOADDETAIL_SDNXTR ASC,
          SBXLOADDETAIL_SANDWEIGHT ASC,
          SBXLOADDETAIL_EXTWEIGHT ASC,
          SBXLOADDETAIL_SHMCU ASC,
          SBXLOADDETAIL_SDDOC ASC,
          SBXLOADDETAIL_GLDATE ASC,
          SBXLOADDETAIL_FRTAMT ASC,
          ID_CUSTOM_89d35e8e9f91315 ASC,
          SBXLOADDETAIL_EXTAMT ASC,
          F554201T_QCLGL1 ASC,
          F554201T_QCLGL2 ASC,
          F554201T_QCLGL3 ASC,
          F554201T_QCFSTR3 ASC,
          SBXLOADDETAIL_HOLAMT ASC,
          SBXLOADDETAIL_SANDAMT ASC,
          F554201T_QCDS50 ASC,
